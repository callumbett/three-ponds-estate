# https://www.robotstxt.org/robotstxt.html
# Allow general crawling of the marketing site so the property gets
# indexed for organic search, but keep the guest-only site-map PDFs
# and Next.js internal paths out of search results.

User-agent: *
Allow: /

# Guest pre-arrival PDFs (linked from check-in email, not navigable
# from the site). Disallow so they don't appear in Google's index.
Disallow: /site-maps/

# Next.js internal paths — image optimisation URLs, build assets, etc.
# Ahrefs was crawling /_next/image?url=…&dpl=… URLs (deployment-tagged
# optimisation endpoints) and flagging them as having too many URL
# parameters. These paths carry no indexable content; disallowing them
# keeps the crawl budget on actual pages and stops the parameter noise.
Disallow: /_next/

# Sitemap location — helps Google, Bing, etc. discover the full
# page list immediately on first crawl.
Sitemap: https://threepondsestate.com/sitemap.xml
